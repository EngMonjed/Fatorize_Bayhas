    old.model_number,
    old.color AS old_color,
    old.barcode AS old_barcode,

    newp.id AS new_product_id,

    ng.group_key,
    ng.packet_qty,
    ng.sizes AS new_sizes,

    pc.id AS new_color_id,
    pc.name AS new_color,

    CASE
        WHEN old.barcode IS NULL OR TRIM(old.barcode) = ''
            THEN 'NO OLD BARCODE'

        WHEN newp.id IS NULL
            THEN 'NO PRODUCT'

        WHEN ng.group_key IS NULL
            THEN 'NO SIZE GROUP'

        WHEN pc.id IS NULL
            THEN 'NO COLOR'

        ELSE 'MATCH'
    END AS match_status

FROM `test`.`products` old

LEFT JOIN `test`.`products_ret` newp
    ON newp.model_number = old.model_number

LEFT JOIN (
    SELECT
        ps.product_id,
        ps.group_key,
        ps.packet_qty,

        MIN(CAST(ps.size AS UNSIGNED)) AS min_size,
        MAX(CAST(ps.size AS UNSIGNED)) AS max_size,
        COUNT(*) AS size_count,

        GROUP_CONCAT(
            ps.size
            ORDER BY CAST(ps.size AS UNSIGNED)
            SEPARATOR ','
        ) AS sizes

    FROM `test`.`product_sizes_ret` ps

    WHERE ps.age_type = 'سنة'
      AND ps.size REGEXP '^[0-9]+$'

    GROUP BY
        ps.product_id,
        ps.group_key,
        ps.packet_qty
) ng

    ON ng.product_id = newp.id

    AND ng.packet_qty = old.packet_size

    AND ng.min_size = old.age_from
    AND ng.max_size = old.age_to

    AND ng.size_count =
        FLOOR(
            (old.age_to - old.age_from) / old.age_step
        ) + 1

LEFT JOIN `test`.`product_colors_ret` pc
    ON pc.name = old.color

ORDER BY
    old.model_number,
    old.age_from,
    old.age_to,
    old.color;

DELIMITER $$
DROP PROCEDURE IF EXISTS drop_branch_tables$$
CREATE PROCEDURE drop_branch_tables(IN branch_suffix VARCHAR(20))
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE tbl_name VARCHAR(255);
    DECLARE cur CURSOR FOR
        SELECT TABLE_NAME FROM information_schema.TABLES
        WHERE TABLE_SCHEMA = DATABASE()
          AND TABLE_NAME LIKE CONCAT('%\\_', branch_suffix);
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
 
    SET FOREIGN_KEY_CHECKS = 0;
 
    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO tbl_name;
        IF done THEN
            LEAVE read_loop;
        END IF;
        SET @sql = CONCAT('DROP TABLE IF EXISTS `', tbl_name, '`');
        PREPARE stmt FROM @sql;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
    END LOOP;
    CLOSE cur;
 
    SET FOREIGN_KEY_CHECKS = 1;
END$$
DELIMITER ;

-- ── خطوة ٣: التنفيذ الفعلي — بدّل 'alpfac' باللاحقة الصح ──
CALL drop_branch_tables('alpfac');
