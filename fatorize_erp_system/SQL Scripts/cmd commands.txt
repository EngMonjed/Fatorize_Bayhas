DELIMITER $$
DROP PROCEDURE IF EXISTS drop_branch_tables$$
CREATE PROCEDURE drop_branch_tables(IN branch_suffix VARCHAR(20))
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE tbl_name VARCHAR(255);
    DECLARE cur CURSOR FOR
        SELECT TABLE_NAME FROM information_schema.TABLES
        WHERE TABLE_SCHEMA = DATABASE()
          AND TABLE_NAME LIKE CONCAT('%\\_', branch_suffix);
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
 
    SET FOREIGN_KEY_CHECKS = 0;
 
    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO tbl_name;
        IF done THEN
            LEAVE read_loop;
        END IF;
        SET @sql = CONCAT('DROP TABLE IF EXISTS `', tbl_name, '`');
        PREPARE stmt FROM @sql;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
    END LOOP;
    CLOSE cur;
 
    SET FOREIGN_KEY_CHECKS = 1;
END$$
DELIMITER ;

-- ── خطوة ٣: التنفيذ الفعلي — بدّل 'alpfac' باللاحقة الصح ──
CALL drop_branch_tables('alpfac');

-- 1_hr_loans_paid_amount.sql
-- غيّر NN لرقم الهجرة التالي الفعلي عندك.
--
-- يضيف عمود `paid_amount` لجداول hr_loans_* بكل الفروع — تتبّع تراكمي
-- للمبلغ الفعلي المسدَّد من كل سلفة، بدل الاعتماد فقط على عدد الأقساط
-- (paid_installments/installments) يلي بيفترض إنه كل خصم = قسط كامل
-- بالضبط. لازم لدعم الخصم الجزئي المخصَّص (مبلغ حر، مو قسط ثابت فقط).
--
-- آمن لإعادة التشغيل: يفحص وجود العمود قبل الإضافة، ويهيّئ القيم
-- الموجودة مسبقاً (paid_amount = paid_installments × monthly_deduction)
-- عشان السلف يلي أصلاً فيها تسديد جزئي من النظام القديم ما تنكشف
-- على إنها صفر بالغلط.

SET @db := DATABASE();

DROP PROCEDURE IF EXISTS hr_loans_paid_amount_migration;

DELIMITER $$
CREATE PROCEDURE hr_loans_paid_amount_migration()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE tbl_name VARCHAR(128);
    DECLARE cur CURSOR FOR
        SELECT TABLE_NAME
        FROM information_schema.TABLES
        WHERE TABLE_SCHEMA = @db AND TABLE_NAME LIKE 'hr_loans\_%';
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN cur;
    loans_loop: LOOP
        FETCH cur INTO tbl_name;
        IF done THEN
            LEAVE loans_loop;
        END IF;

        IF NOT EXISTS (SELECT 1 FROM information_schema.COLUMNS
            WHERE TABLE_SCHEMA=@db AND TABLE_NAME=tbl_name AND COLUMN_NAME='paid_amount') THEN
            SET @sql = CONCAT('ALTER TABLE `', tbl_name, '` ADD COLUMN `paid_amount` DECIMAL(15,2) NOT NULL DEFAULT 0 AFTER `paid_installments`');
            PREPARE stmt FROM @sql; EXECUTE stmt; DEALLOCATE PREPARE stmt;

            -- تهيئة القيم الموجودة مسبقاً بناءً على عدد الأقساط المسجّلة
            -- (أفضل تقدير متاح للبيانات القديمة، مش افتراض صفر أعمى)
            SET @sql = CONCAT('UPDATE `', tbl_name, '` SET paid_amount = paid_installments * monthly_deduction WHERE paid_amount = 0 AND paid_installments > 0');
            PREPARE stmt FROM @sql; EXECUTE stmt; DEALLOCATE PREPARE stmt;

            SELECT CONCAT('تم تحديث: ', tbl_name) AS log;
        END IF;
    END LOOP;
    CLOSE cur;
END$$
DELIMITER ;

CALL hr_loans_paid_amount_migration();
DROP PROCEDURE hr_loans_paid_amount_migration;
